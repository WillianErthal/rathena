-	script	::PR_Officer_moscovia	-1,{
        mes "[Moscovia P.R. Officer]";
        mes "Moscovia! Você conhece Moscovia?";
        mes "O paraíso que se estende pelos mares sem fim...";
        mes "Bem-vindo(a) a Moscovia";
        mes "Ela é aventureira e mística.";
        next;
        switch(select("Sobre Moscovia...:Ir para Moscovia.:Cancelar")) {
		case 1:
			mes "[Moscovia P.R. Officer]";
                        mes "Moscovia é um reino muito bonito";
                        mes "em uma ilha localizada ao norte de";
                        mes "Rune-Midgarts.";
                        next;
                        mes "[Moscovia P.R. Officer]";
                        mes "Tenho certeza de que você ficará";
                        mes "absolutamente fascinado";
                        mes "pelas belas paisagens de Moscovia";
                        mes "e por seu magnífico palácio.";
                        next;
                        mes "[Moscovia P.R. Officer]";
                        mes "Agora que nosso longo inverno";
                        mes "passou,";
                        mes "fico feliz por poder mostrar a você";
                        mes "nossa magnífica cidade natal.";
                        next;
                        mes "[Moscovia P.R. Officer]";
                        mes "Se quiser visitar Moscovia,";
                        mes "aproveite esta chance!";
                        mes "Vou ajudar você a ter uma ótima";
                        mes "viagem para Moscovia!";
                        close;

		case 2:
                        mes "[Moscovia P.R. Officer]";
                        mes "Certo, vamos começar agora.";
                        mes "Você deve me pagar 10.000 zeny";
                        mes "para ir a Moscovia.";
                        mes "Mas quando voltar,";
                        mes "não terá que pagar.";
                        next;
                        mes "[Moscovia P.R. Officer]";
                        mes "Podemos partir agora?";
                        next;
                        if (select("Vamos lá!:Cancelar") == 2) {
                                mes "[Moscovia P.R. Officer]";
                                mes "Se estiver ocupado demais agora,";
                                mes "fale comigo quando quiser.";
                                mes "Estou sempre pronto para guiar qualquer um a Moscovia.";
                                close;
                        }
                        if (Zeny < 10000) {
                                mes "[Moscovia P.R. Officer]";
                                mes "Sinto muito, mas você não tem";
                                mes "zeny suficiente agora.";
                                mes "Você precisa de 10.000 zeny";
                                mes "para ir a Moscovia.";
                                mes "Obrigado.";
                                close;
                        } else {
                                mes "[Moscovia P.R. Officer]";
                                mes "Certo, estamos partindo agora.";
                                close2;
                                set Zeny,Zeny - 10000;
                                warp "moscovia",163,55;
				end;
			}
                case 3:
                        mes "[Moscovia P.R. Officer]";
                        mes "Se estiver ocupado demais agora,";
                        mes "fale comigo novamente quando quiser.";
                        mes "Estou sempre pronto para guiar qualquer um até Moscovia.";
                        close;
        }
}
